%-----------------------------------------------------
% COLOR PALETTE — Aqua, Cyan, Glassy Whites
%-----------------------------------------------------
\definecolor{aerobg}{HTML}{E0F7FA}        % Soft aqua background
\definecolor{aerobgdeep}{HTML}{B2EBF2}    % Deeper pool blue
\definecolor{aerocyan}{HTML}{00BCD4}      % Vibrant cyan
\definecolor{aerodark}{HTML}{006064}      % Deep teal
\definecolor{aeroglass}{HTML}{FFFFFF}     % Pure white for glass
\definecolor{aeroglow}{HTML}{4DD0E1}      % Soft glow cyan
\definecolor{aerosky}{HTML}{81D4FA}       % Sky blue
\definecolor{aerogreen}{HTML}{69F0AE}     % Mint green accent

%-----------------------------------------------------
% BEAMER TEMPLATE CONFIGURATION
%-----------------------------------------------------
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}[frame number]
\setbeamertemplate{frametitle}{%
    \vspace{0.4cm}%
    \begin{tikzpicture}[remember picture, overlay]
        % Glossy title bar background
        \fill[left color=aerocyan, right color=aerosky, middle color=aeroglow, opacity=0.85]
            (0,0) rectangle (\paperwidth, 1.2cm);
        % Highlight strip (glass reflection)
        \fill[aeroglass, opacity=0.35]
            (0,0.75cm) rectangle (\paperwidth, 1.05cm);
        % Title text with subtle shadow
        \node[anchor=west, text=white, font=\Large\bfseries, 
              drop shadow={shadow xshift=0.5pt, shadow yshift=-0.5pt, opacity=0.3}]
            at (0.5cm, 0.6cm) {\insertframetitle};
    \end{tikzpicture}%
    \vspace{0.9cm}%
}

% CUSTOM BLOCKS — Glass Morphism Style
%-----------------------------------------------------
\setbeamertemplate{block begin}{%
    \begin{tikzpicture}
        \node[rounded corners=8pt, 
              fill=aeroglass, fill opacity=0.65,

              blur shadow={shadow blur steps=5, shadow xshift=1pt, shadow yshift=-1pt, shadow opacity=0.15}]

              \vspace{2pt}
              \end{minipage}
              \egroup;
    \end{tikzpicture}
    \vspace{0.5cm}
}

    \begin{tikzpicture}
        \node[rounded corners=8pt,

              inner sep=10pt,
              blur shadow={shadow blur steps=5, shadow xshift=1pt, shadow yshift=-1pt, shadow opacity=0.2}]
              \bgroup
              \begin{minipage}{0.95\textwidth}
              \vspace{2pt}
              {\color{aerodark}\bfseries\large \insertblocktitle}\vspace{4pt}\par

\setbeamertemplate{block alerted end}{%
              \vspace{2pt}
              \end{minipage}

    \end{tikzpicture}
    \vspace{0.5cm}
}

%-----------------------------------------------------
% BACKGROUND — Glossy Aqua Gradient with "Light" Effects
%-----------------------------------------------------
\setbeamertemplate{background}{%
    \begin{tikzpicture}[remember picture, overlay]
        % Base gradient (soft pool effect)
        \shade[top color=aerobg, bottom color=aerobgdeep]
            (current page.north west) rectangle (current page.south east);
        
        % Soft ambient orbs (glows)
        \fill[aeroglow, opacity=0.15] 
            ($(current page.north west)+(2cm,-3cm)$) circle (4cm);
        \fill[aerosky, opacity=0.12] 
            ($(current page.south east)+(-4cm,3cm)$) circle (5cm);
        \fill[aerogreen, opacity=0.08] 
            ($(current page.north east)+(-3cm,-5cm)$) circle (3.5cm);
        
        % Top "glass" reflection bar
        \shade[top color=aeroglass, bottom color=aeroglass, opacity=0.2]
            (current page.north west) rectangle ($(current page.north east)+(0,-2cm)$);
        \draw[aeroglass, opacity=0.4, line width=1pt]
            ($(current page.north west)+(0,-2cm)$) -- ($(current page.north east)+(0,-2cm)$);
            
        % Bottom subtle wave accent
        \fill[aerocyan, opacity=0.1] 
            (current page.south west) .. controls ++(3cm,1cm) and ++(-3cm,0cm) .. ++(8cm,0.8cm)
            .. controls ++(3cm,0cm) and ++(-3cm,0.5cm) .. (current page.south east)
            -- (current page.south east) -- (current page.south west) -- cycle;
    \end{tikzpicture}%
}

    \begin{tikzpicture}[remember picture, overlay]
        % Large glossy card for title
        \node[rounded corners=15pt,
              fill=aeroglass, fill opacity=0.75,
              draw=aerocyan, line width=1.5pt,
              blur shadow={shadow blur steps=8, shadow xshift=2pt, shadow yshift=-2pt, shadow opacity=0.2},
              text width=0.8\paperwidth, align=center,
              inner sep=20pt]
            at (current page.center) {%
                {\color{aerodark}\Huge\bfseries\inserttitle}\\[0.4cm]
                {\color{aerocyan}\large\insertsubtitle}\\[0.8cm]
                {\color{aerodark}\normalsize\insertauthor}\\[0.2cm]
                {\color{aerodark}\small\insertinstitute}\\[0.2cm]
                {\color{aerodark}\footnotesize\insertdate}
            };
        
        % Decorative glossy orb behind title
        \fill[aerocyan, opacity=0.2] 
            ($(current page.center)+(-4cm,2.5cm)$) circle (2.5cm);
        \fill[aeroglass, opacity=0.3] 
            ($(current page.center)+(-4cm,2.5cm)+(0.3cm,0.3cm)$) circle (1.8cm);
    \end{tikzpicture}%
}
