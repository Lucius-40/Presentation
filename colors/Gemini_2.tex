% ==========================================
\definecolor{FaCyan}{RGB}{140, 230, 255}
\definecolor{FaBlue}{RGB}{20, 110, 220}
\definecolor{FaDarkBlue}{RGB}{10, 50, 120}
\definecolor{FaGreen}{RGB}{140, 210, 60}
\definecolor{FaGlass}{RGB}{255, 255, 255}

% ==========================================
% 2. GLOBAL AERO BACKGROUND THEME
% ==========================================
\setbeamertemplate{navigation symbols}{} % Remove default symbols
\setbeamertemplate{background canvas}{
    \begin{tikzpicture}[remember picture, overlay]
        % Base Gradient (Deep blue to light cyan)
        \shade[top color=FaCyan!60, bottom color=FaBlue!80] 
            (current page.south west) rectangle (current page.north east);
            
        % Aurora Swoosh 1 (Vibrant Greenish-blue curve)
        \fill[FaGreen!40, opacity=0.3] 
            (current page.south west) .. controls +(right:6cm) and +(down:4cm) .. 
            ([xshift=-2cm]current page.north east) -- (current page.north east) -- 
            (current page.south east) -- cycle;
            
        % Aurora Swoosh 2 (Glossy white overlay for glass reflection)
        \fill[white, opacity=0.25] 
            (current page.south west) .. controls +(up:6cm) and +(left:7cm) .. 
            (current page.north east) -- (current page.north west) -- cycle;

        % Reusable Bubble Macro in Background
        \foreach \x/\y/\r in {1.5/2/0.6, 12/7/0.9, 4/6.5/0.35, 14/1.5/0.7, 7/1/0.2} {
            % Outer bubble edge
            \draw[white, line width=1pt, opacity=0.5] 
                ($(current page.south west) + (\x,\y)$) circle (\r);
            % Inner glass fill
            \fill[white, opacity=0.15] 
                ($(current page.south west) + (\x,\y)$) circle (\r);
            % Bubble specular highlight (glossy dot)
            \fill[white, opacity=0.65] 
                ($(current page.south west) + (\x-0.25*\r, \y+0.35*\r)$) circle (0.15*\r);
        }
    \end{tikzpicture}
}

% Command: \FaBlock{Title}{Content}
% Creates a frosted glass "Aero" window block with a glossy top reflection.
\newcommand{\FaBlock}[2]{
    \vspace{0.2cm}
    \noindent
    \begin{tikzpicture}
        % Main frosted glass background
        \node[draw=white, line width=1.5pt, rounded corners=2ex, 
              top color=white!95!FaCyan, bottom color=white!60!FaCyan, 
              inner sep=12pt, text width=\textwidth-24pt, align=left, 
              drop shadow={opacity=0.3, shadow xshift=0pt, shadow yshift=-2pt}] (box) {
            {\Large\bfseries\color{FaDarkBlue} #1}\\[1.5ex]
            \color{black!85}\large #2
        };
        
        % Glossy top reflection (Aero Glass effect)
        \begin{scope}
            \clip[rounded corners=2ex] (box.south west) rectangle (box.north east);
            \fill[white, opacity=0.45] 
                (box.north west) -- (box.north east) -- 
                ([yshift=-0.7cm]box.north east) .. controls ([yshift=-1.1cm]box.center) .. 
                ([yshift=-0.4cm]box.north west) -- cycle;
        \end{scope}
    \end{tikzpicture}
    \vspace{0.2cm}
}

% Command: \FaButton{Text}
% Creates a glossy 2006-era green confirmation button.
\newcommand{\FaButton}[1]{
    \tikz[baseline=-0.6ex]{
        \node[draw=white, line width=1pt, rounded corners=1.5ex,
              top color=FaGreen!90!white, bottom color=FaGreen!60!black,
              text=white, font=\bfseries\large, inner sep=5pt, inner xsep=12pt,
              drop shadow={opacity=0.4, shadow xshift=0pt, shadow yshift=-1pt}] (btn) {#1};
        % Glossy button highlight
        \begin{scope}
            \clip[rounded corners=1.5ex] (btn.south west) rectangle (btn.north east);
            \fill[white, opacity=0.35] 
                (btn.north west) rectangle ([yshift=-1.4ex]btn.north east);
        \end{scope}
    }
}
