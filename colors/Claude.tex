% --- Reusable: a "glass pill" bar with a glossy highlight ---
% args: #1 = width, #2 = height, #3 = base color
\newcommand{\aeroGlassPill}[3]{%
  \begin{tikzpicture}
    \fill[rounded corners=0.18cm,
          top color=#3!55!white, bottom color=#3!85!black]
      (0,0) rectangle (#1,#2);
    \fill[rounded corners=0.16cm, white, opacity=0.55]
      (0.06,#2*0.5) rectangle (#1-0.06,#2-0.06);
    \fill[rounded corners=0.12cm, white, opacity=0.28]
      (0.10,#2*0.55) rectangle (#1-0.10,#2-0.12);
  \end{tikzpicture}%
}

% --- Floating bubble accents used on backgrounds ---
\newcommand{\aeroBubble}[3]{% x, y, radius
  \shade[shading=radial, inner color=white, outer color=aeroSkyMid,
         opacity=0.35] (#1,#2) circle (#3);
  \shade[shading=radial, inner color=white, outer color=white,
         opacity=0.0] (#1,#2) circle (#3);
  \fill[white, opacity=0.55] ($(#1,#2)+(-#3*0.35,#3*0.35)$) circle (#3*0.22);
}
