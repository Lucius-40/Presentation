\documentclass[aspectratio=169]{beamer}
\usepackage[utf8]{inputenc}
\usepackage{tikz}
\usepackage{tcolorbox}

% --- Frutiger Aero Color Palette ---
\definecolor{AeroSky}{RGB}{14, 165, 233}       % Vibrant sky blue
\definecolor{AeroDeep}{RGB}{2, 48, 71}         % Deep water contrast blue
\definecolor{AeroEco}{RGB}{34, 197, 94}        % Aurora/nature green
\definecolor{AeroGlass}{RGB}{240, 249, 255}    % Glossy base tint
\definecolor{AeroText}{RGB}{15, 23, 42}        % High-contrast dark text

% --- Typography ---
\usefonttheme{professionalfonts}
\renewcommand{\familydefault}{\sfdefault}

% --- Global Theme Elements ---
\setbeamercolor{structure}{fg=AeroSky}
\setbeamercolor{normal text}{fg=AeroText}
\setbeamercolor{title}{fg=white}
\setbeamercolor{frametitle}{fg=white}

% Disable standard navigation symbols
\setbeamertemplate{navigation symbols}{}

% --- Custom Dynamic Glass Background ---
\setbeamertemplate{background}{
    \begin{tikzpicture}[remember picture, overlay]
        % Glossy Sky-to-Water Gradient
        \shadedraw [top color=AeroSky!90!white, bottom color=AeroSky!40!AeroEco!20!white, draw=none] 
            (current picture.north west) rectangle (current picture.south east);
        
        % Abstract Aurora/Glass Wave (Skeuomorphic Curve)
        \fill[white, opacity=0.15] 
            (current picture.south west) to[out=45,in=225] (current picture.east) 
            -- (current picture.north east) -- (current picture.north west) -- cycle;
            
        % Subtle Eco-Bubble Accents
        \fill[AeroEco, opacity=0.12] (current picture.south east) ++(-2,-0.5) circle (3cm);
        \fill[white, opacity=0.08] (current picture.center) ++(-4,2) circle (1.5cm);
    \end{tikzpicture}
}

% --- Frutiger Aero "Glass" Blocks ---
\tcbuselibrary{skins}
\newtcolorbox{aeroblock}[1]{
    colback=white!60!AeroGlass,
    colframe=white,
    coltitle=white,
    fonttitle=\bfseries,
    colbacktitle=AeroSky!85!AeroDeep,
    enhanced,
    attach boxed title to top left={yshift=-2mm, xshift=3mm},
    boxed title style={
        sharp corners=downhill, 
        arc=3mm, 
        colframe=white!80!AeroSky, 
        interior style={top color=AeroSky, bottom color=AeroSky!70!AeroDeep}
    },
    arc=4mm,
    boxrule=1.5pt,
    shadow={0.5mm}{-0.5mm}{0mm}{black!10}, % Subtle depth dropping
    title=#1
}
